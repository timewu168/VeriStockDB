Private Sub Workbook_Open()
    Dim strSelect As String
    AppStop
    Sheets("SetUp").[C4].Value = Date '輸入最新日期
    If Day(Date) > 10 Then '如果在10號之後
        Sheets("SetUp").[C17].Value = DateAdd("m", -1, Date - Day(Date) + 1)
    ElseIf Day(Date) <= 10 Then '如果在10號之前
        Sheets("SetUp").[C17].Value = DateAdd("m", -2, Date - Day(Date) + 1)
    End If
    Sheets("TradingDay").Select '作用工作表移到TradingDay
    Sheets("TradingDay").Cells(2, "A").Resize(10000, 1).ClearContents
    Call GetCheckData("", "交易日", 1, "HTML", "UTF-8", 1) '呼叫爬蟲取得期交所最新交易日
    Range("A2:A31").Sort key1:=Range("A2:A31"), Order1:=1, Header:=xlNo '排序，最舊到最新
    '時間判定，晚上九點前不更新最新交易日
    If Weekday(Date) = 1 Or Weekday(Date) = 7 Then '判斷是否是星期六日
        Sheets("TradingDay").[A1048576].End(xlUp).ClearContents
    ElseIf Time < TimeValue("15:30:00") Then '如果是星期一到五的下午15:30前
        Sheets("TradingDay").[A1048576].End(xlUp).ClearContents
    End If
    Sheets("SetUp").Cells(5, "C").Value = Sheets("TradingDay").[A1048576].End(xlUp).Value '最新交易日=交易日期最後一列的值
    '=====連結資料庫取得各資料表最新日期=====
    strSelect = "SELECT 交易日期 FROM Close order by 交易日期 desc LIMIT 1" 'select 日收盤資料庫最新日期
    Call SQLiteDB(strSelect, "SetUp", 6, "C") '取得日收盤最新日期
    Call GetYesterDayData '呼叫取得最後交易日收盤資料程式
    strSelect = "SELECT 交易日期 FROM Legal order by 交易日期 desc LIMIT 1" 'select 日法人資料庫最新日期
    Call SQLiteDB(strSelect, "SetUp", 7, "C") '取得日法人最新日期
    strSelect = "SELECT 交易日期 FROM Margin order by 交易日期 desc LIMIT 1" 'select 日資券資料庫最新日期
    Call SQLiteDB(strSelect, "SetUp", 8, "C") '取得日資券最新日期
    strSelect = "SELECT 交易日期 FROM DayTrading order by 交易日期 desc LIMIT 1" 'select 日當沖資料庫最新日期
    Call SQLiteDB(strSelect, "SetUp", 9, "C") '取得日當沖最新日期
    strSelect = "SELECT 交易日期 FROM Revenue order by 交易日期 desc LIMIT 1" 'select 月營收資料庫最新日期
    Call SQLiteDB(strSelect, "SetUp", 18, "C") '取得日當沖最新日期
    strSelect = "SELECT 公告日期 FROM Notice order by 公告日期 desc LIMIT 1" 'select 注意資料庫最新日期
    Call SQLiteDB(strSelect, "SetUp", 10, "C") '取得日當沖最新日期
    strSelect = "SELECT 公告日期 FROM Disposal WHERE 市場別 = '上市' order by 公告日期 desc LIMIT 1" 'select 上市處資資料庫最新日期
    Call SQLiteDB(strSelect, "SetUp", 11, "C") '取得日當沖最新日期
    strSelect = "SELECT 公告日期 FROM Disposal WHERE 市場別 = '上櫃' order by 公告日期 desc LIMIT 1" 'select 上櫃處置資料庫最新日期
    Call SQLiteDB(strSelect, "SetUp", 12, "C") '取得日當沖最新日期
    Sheets("SetUP").Select
    AppStart
End Sub

Public Function CorrectionCode(strCode As String)
    If Len(strCode) = 3 Then '如果證券代號只有3碼
        CorrectionCode = "00" & CStr(strCode) '股票代號"
    ElseIf strCode = "6201" Then  '如果是元大富櫃50
        CorrectionCode = "00" & CStr(strCode) '股票代號"
    ElseIf Len(strCode) = 5 And Mid(strCode, 1, 1) = "2" Then '如果證券代號只有5碼，而且第一碼=2
        CorrectionCode = "0" & CStr(strCode) '股票代號"
    Else
        CorrectionCode = CStr(strCode) '股票代號"
    End If
End Function

Public Function MyPath(strSheets As String) As String
    Select Case strSheets
        Case "File"
            MyPath = ThisWorkbook.Path & "\"
        Case "csv"
            MyPath = ThisWorkbook.Path & "\csv\"
        Case "Data"
            MyPath = ThisWorkbook.Path & "\Data\"
        Case "SQLite"
            MyPath = ThisWorkbook.Path & "\SQLite\"
        Case "txt"
            MyPath = ThisWorkbook.Path & "\txt\"
        Case "DLF"
            MyPath = "C:\Users\" & Environ$("username") & "\Downloads\"
    End Select
End Function

Public Function Code2Row(strCode, cRange) As String
    On Error Resume Next
    Dim strRow As String
    Dim rngCode As Range
    Select Case cRange
        Case "Close"
            Set rngCode = Sheets("TradingDay").Range("C:C")
        Case "ChCloseS"
            Set rngCode = Sheets("CheckData").Range("A:A")
        Case "ChCloseO"
            Set rngCode = Sheets("CheckData").Range("H:H")
    End Select
    If WorksheetFunction.CountIf(rngCode, strCode) > 0 Then
        strRow = WorksheetFunction.Match(strCode, rngCode, 0)
    End If
    If strRow <> "" Then
        Code2Row = strRow
    Else
        Code2Row = 0
    End If
End Function

Function convertraw(rawdata, char)
    Dim rawstr
    Set rawstr = CreateObject("adodb.stream")
    With rawstr
        .Type = 1
        .Mode = 3
        .Open
        .Write rawdata
        .Position = 0
        .Type = 2
        .Charset = char
        convertraw = .ReadText
        .Close
    End With
    Set rawstr = Nothing
End Function

Public Function FunCheckData(FNameM As String, DBDay As Date)
    Dim strC As String, strF As String
    Select Case FNameM
        Case "CloseSII"
            strC = "A"
            strF = "E"
        Case "CloseOTC"
            strC = "H"
            strF = "L"
        Case "LegalSII"
            strC = "O"
            strF = "P"
        Case "LegalOTC"
            strC = "AA"
            strF = "AB"
        Case "MarginSII"
            strC = "AM"
            strF = "AN"
        Case "MarginOTC"
            strC = "BC"
            strF = "BD"
        Case "DayTradingSII"
            strC = "BS"
            strF = "CA"
        Case "DayTradingOTC"
            strC = "CJ"
            strF = "CR"
        Case "RevenueSII"
            strC = "DB"
            strF = "DC"
        Case "RevenueOTC"
            strC = "DH"
            strF = "DI"
    End Select
    i = 2
    If Mid(FNameM, 1, 5) = "Close" Or Mid(FNameM, 1, 5) = "DayTr" Then
        FileDay = DBDay
    ElseIf Mid(FNameM, 1, 5) = "Reven" Then
        FileDay = Format(CStr(DateAdd("m", 1, DBDay - Day(DBDay) + 1)), "yyyy") & "/" & Format(CStr(DateAdd("m", 1, DBDay - Day(DBDay) + 1)), "mm")  '取得下個月月營收日期格式
        For d = 7 To 54
            If Mid(Sheets("CheckData").Cells(d, strC), 1, 1) = "1" Then
                Sheets("CheckData").Cells(d, strC) = Mid(Sheets("CheckData").Cells(d, strC), 1, 3) + 1911 & Mid(Sheets("CheckData").Cells(d, strC), 4, 3) & "/01"
            End If
        Next
    Else
        FileDay = Year(DBDay) - 1911 & "/" & Month(DBDay) & "/" & Day(DBDay)
    End If
    Do While Sheets("CheckData").Cells(i, strC) <> FileDay
        If i > 500 Then
            MsgBox ("找不到交易日")
            End
        End If
        i = i + 1
        cRow = i
    Loop
    FunCheckData = Sheets("CheckData").Cells(i, strF)
End Function

'=====數據清理程式=====
Sub AllMergeData(strSheets As String)
    Dim myFileName As String  ' 設定檔案名稱變數
    Dim strMarket As String '設定市場變數
    Dim MyFile As Object '設定檔案物件變數
    Sheets("MergeData").Select     '作用中的工作表轉移到MergeData
    Columns("B:B").NumberFormatLocal = "@" 'b欄位為文字
    myFileName = Dir(MyPath("csv") & "*") '取得檔案名稱
    Do While myFileName <> "" '檔名不為空白
        If strSheets = "Revenue" Then '如果是月營收
            FileDay = Mid(myFileName, 1, 4) & "/" & Mid(myFileName, 5, 2) & "/" & "01" '取得日期並改格式
        Else '如果不是月營收
            FileDay = Mid(myFileName, 1, 4) & "/" & Mid(myFileName, 5, 2) & "/" & Mid(myFileName, 7, 2) '取得日期並改格式
        End If
        Select Case strSheets
            Case "Close"
                DBDay = DateValue(Sheets("SetUp").Cells(6, "C"))
            Case "Legal"
                DBDay = DateValue(Sheets("SetUp").Cells(7, "C"))
            Case "Margin"
                DBDay = DateValue(Sheets("SetUp").Cells(8, "C"))
            Case "DayTrading"
                DBDay = DateValue(Sheets("SetUp").Cells(9, "C"))
            Case "Revenue"
                DBDay = DateValue(Sheets("SetUp").Cells(18, "C"))
            Case "Notice"
                DBDay = DateValue(Sheets("SetUp").Cells(10, "C"))
            Case "Disposal"
                DBDay = DateValue(Sheets("SetUp").Cells(11, "C"))
        End Select
        Application.StatusBar = "目前進度:" & myFileName '顯示進度
        DoEvents '回應使用者操作事件
        If strSheets = "Revenue" And DateValue(FileDay) > DateValue(DBDay) And DateValue(FileDay) <= DateValue(Sheets("SetUp").Cells(17, "C")) Then GoTo startRevenue '跳到開始處理月營收資料
        If Right(myFileName, 3) = "csv" And Mid(myFileName, 9, 5) = Mid(strSheets, 1, 5) And DateValue(FileDay) > DateValue(DBDay) And DateValue(FileDay) <= DateValue(Sheets("SetUp").Cells(5, "C")) Then ' 判斷檔案格式及日期
startRevenue: '開始處理月營收資料
            strMarket = Mid(Right(myFileName, 7), 1, 3) '取得市場別
            With ActiveSheet.QueryTables.Add("TEXT;" & MyPath("csv") & myFileName, Sheets("MergeData").[$T$1])
                .TextFileParseType = xlDelimited '文件解析類型為使用分隔符分割
                If strSheets = "Revenue" Then .TextFilePlatform = 65001 'UTF-8
                .TextFileTabDelimiter = False '取消tab鍵
                .TextFileCommaDelimiter = True '使用","
                .Refresh BackgroundQuery:=False '是否背景執行
                Sheets("MergeData").Range("S1") = FileDay '輸入FileDay給處理資料的程式使用
                Call SheetsMarket(strSheets & strMarket)
                .ResultRange.Clear '把處理好的資料刪掉
                .Delete '把連線及名稱刪掉
                Sheets("MergeData").Range("S1").ClearContents '刪掉FileDay
            End With
            '移動處理好的csv
            Set MyFile = CreateObject("Scripting.FileSystemObject")
            MyFile.MoveFile MyPath("csv") & myFileName, MyPath("csv") & strSheets & "\"
            Set MyFile = Nothing
        End If
        myFileName = Dir '取得下一個檔名
        If strSheets = "Revenue" And myFileName <> "" Then '如果還有月營收檔案
            FileDay = Mid(myFileName, 1, 4) & "/" & Mid(myFileName, 5, 2) & "/" & "01" '取得日期並改格式
            GoTo startRevenue '繼續處理資料
        ElseIf strSheets = "Revenue" And myFileName = "" Then '如果是最後一個月營收檔案
            GoTo RevenueUpDB '把處理好的資料上傳資料庫
        End If
        If FileDay <> Mid(myFileName, 1, 4) & "/" & Mid(myFileName, 5, 2) & "/" & Mid(myFileName, 7, 2) And DateValue(FileDay) > DateValue(DBDay) And DateValue(FileDay) <= DateValue(Sheets("SetUp").Cells(5, "C")) Then  '如果下一個檔名的日期<>FileDay，就開始處理整理好的資料
RevenueUpDB: '把處理好的資料上傳資料庫
            Call InputTitle(strSheets) '輸入Title
            StartRow = Sheets("MergeData").[A1048576].End(xlUp).Row '取得最後一列
            strColumn = Sheets("MergeData").[A1].End(xlToRight).Column  '取得最後一欄
            EndRow = Sheets("MergeData").Cells(1, CInt(strColumn) - 1).End(xlDown).Row '取得最後一欄的最後一列，檢查用
            If strSheets = "Disposal" And StartRow = 1 Then '如果當天沒資料
                Sheets("SetUp").Select     '作用中的工作表轉移到SetUp
                End '就結束程式
            End If
            If StartRow = EndRow Then
                Sheets.Add After:=Sheets(Sheets.Count)  ' 新增最後1張工作表
                Sheets(Sheets.Count).Select       ' 選取最後一張工作表
                Sheets(Sheets.Count).Name = strSheets '工作表改名
                Columns("B:B").NumberFormatLocal = "@" 'b欄位為文字
                Columns("A:A").NumberFormatLocal = "yyyy-MM-dd" '設定為這個格式，日期才能正確排序
                Sheets(strSheets).[A1].Resize(StartRow, strColumn).Value = Sheets("MergeData").[A1].Resize(StartRow, strColumn).Value '把整理好的資料移到新的工作表
                Sheets(strSheets).Move '移動工作表至新的活頁簿
                '另存為csv_UTF8格式，但副檔名要改成txt，這樣子股票代號才會完整，而且中文股名才會顯示出來
                ActiveWorkbook.SaveAs Filename:="" & MyPath("SQLite") & "All" & strSheets & ".txt", _
                    FileFormat:=xlCSVUTF8, CreateBackup:=False
                ActiveWindow.Close '關掉作用中的活頁簿
                Sheets("MergeData").Select '作用中的工作表轉移到MergeData
                Cells.Select '全選工作表
                Selection.Delete Shift:=xlUp '刪除(不能用ClearContents，有時會留下""資料欄位)
                '上傳資料庫
                fS = FileLen(MyPath("Data") & "Stock.db")  '取得db的容量
                Script = Shell(MyPath("SQLite") & "SQLite.bat " & MyPath("Data") & "stock.db " & strSheets, 0)  '上傳資料庫
                ss = 1
                If strSheets <> "Disposal" Then '因處置資料量少，所以跳過不檢查(或是改成檢查日期，這部份再想想)
                    Do While FileLen(MyPath("Data") & "Stock.db") = fS
                        '如果容量沒改變，代表還沒上傳完成，所以等待1秒再判斷容量是否改變了
                        Application.Wait (Now + TimeValue("0:00:01"))  '設定等待秒數
                        ss = ss + 1
                        If ss = 5 Then MsgBox "ERROR"
                        DoEvents '回應使用者操作事件
                    Loop
                End If
                If strSheets = "Close" Then Sheets("SetUp").Cells(6, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "Legal" Then Sheets("SetUp").Cells(7, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "Margin" Then Sheets("SetUp").Cells(8, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "DayTrading" Then Sheets("SetUp").Cells(9, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "Notice" Then Sheets("SetUp").Cells(10, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "Disposal" Then
                    Sheets("SetUp").Cells(11, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                    Application.Wait (Now + TimeValue("0:00:01"))  '設定等待秒數，如果是處置資料，這邊要等一秒，資料才會完整
                End If
                If strSheets = "Close" Then Sheets("TradingDay").[A1048576].End(xlUp).Offset(1, 0).Value = FileDay '更新交易日日期(改成只有close要做)
                Kill MyPath("SQLite") & "All" & strSheets & ".txt" '如果a15的值等於空白就刪掉剛剛的檔案"
                If strSheets = "Close" Then Call GetYesterDayData '取得最新資料庫日期的資料
            Else
                MsgBox ("資料有缺，請檢查原始資料")
                End
            End If
        End If
    Loop
    Sheets("SetUp").Select     '作用中的工作表轉移到SetUp
End Sub

'月營收
Sub RevenueMerge()
        '選取範圍並復制到B欄
        StartRow = Evaluate("MATCH(""出表日期"",T:T,0)+1")
        EndRow = [W1048576].End(xlUp).Row
        strColumn = Sheets("MergeData").Cells(StartRow - 1, "T").End(xlToRight).Column - 2 '取得最後一欄
        Call CheckCsvTitle("Revenue", StartRow - 1, CDate(Sheets("MergeData").[S1].Value)) '檢查Title
        C = EndRow - StartRow
        m = Range("A1048576").End(xlUp).Offset(1, 0).Row
        e = m + C
        Range("A" & m & ":A" & e).Value = CDate(Sheets("MergeData").[S1].Value) '日期
        Range("B" & m & ":E" & e).Value = Range("V" & StartRow & ":Y" & EndRow).Value '代號、名稱、產業別、當月營收
        Range("F" & m & ":F" & e).Value = Range("AG" & StartRow & ":AG" & EndRow).Value '備註
        For i = m To e
            Sheets("MergeData").Cells(i, "B").Value = CorrectionCode(Trim(Sheets("MergeData").Cells(i, "B").Text)) '處理股票代號
            Sheets("MergeData").Cells(i, "C").Value = Trim(Sheets("MergeData").Cells(i, "C").Text) '處理股票名
        Next
End Sub

'日當沖
Sub DayTradingMerge()
        '選取範圍並復制到B欄
        StartRow = Evaluate("MATCH(""證券代號"",T:T,0)+1")
        EndRow = [W1048576].End(xlUp).Row
        strColumn = Sheets("MergeData").Cells(StartRow - 1, "T").End(xlToRight).Column - 19 '取得最後一欄
        Call CheckCsvTitle("DayTrading", StartRow - 1, CDate(Sheets("MergeData").[S1].Value)) '檢查Title
        C = EndRow - StartRow
        m = Range("A1048576").End(xlUp).Offset(1, 0).Row
        e = m + C
        Range("A" & m & ":A" & e).Value = CDate(Sheets("MergeData").[S1].Value) '日期
        '<= 2014/06/27 otc少一欄，>=2014/06/30都一樣，要處理一下
        If strColumn = 6 Then
            Range("B" & m & ":G" & e).Value = Range("T" & StartRow & ":Y" & EndRow).Value '所有資料
        Else
            Range("B" & m & ":C" & e).Value = Range("T" & StartRow & ":U" & EndRow).Value '所有資料
            Range("E" & m & ":G" & e).Value = Range("V" & StartRow & ":X" & EndRow).Value '所有資料
        End If
        For i = m To e
            Sheets("MergeData").Cells(i, "B").Value = CorrectionCode(Trim(Sheets("MergeData").Cells(i, "B").Text)) '處理股票代號
            Sheets("MergeData").Cells(i, "C").Value = Trim(Sheets("MergeData").Cells(i, "C").Text) '處理股票名
        Next
End Sub

'上市收盤
Sub CloseSIIMerge()
    StartRow = Evaluate("MATCH(""證券代號"",T:T,0)+1")
    EndRow = [U1048576].End(xlUp).Row
    Call CheckCsvTitle("CloseSII", StartRow - 1, CDate(Sheets("MergeData").[S1].Value))
    For Z = StartRow To EndRow
        Sheets("MergeData").[A1048576].End(xlUp).Offset(1, 0).Value = CDate(Sheets("MergeData").[S1].Value) '日期
        Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 1).Value = Trim(CStr(Sheets("MergeData").Cells(Z, "T").Text)) '股票代號
        strCode = Cells(Z, "T").Text '股票代號變數
        Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 2).Value = Trim(Sheets("MergeData").Cells(Z, "U").Text) '股票名
        cRow = Code2Row(strCode, "Close")
        If Cells(Z, "Y").Value = "--" Then
            If cRow > 0 Then
                Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 3).Value = Sheets("TradingDay").Cells(cRow, "H").Value '改h才對
                Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 4).Value = Sheets("TradingDay").Cells(cRow, "H").Value '改h才對
                Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 5).Value = Sheets("TradingDay").Cells(cRow, "H").Value '改h才對
                Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 6).Value = Sheets("TradingDay").Cells(cRow, "H").Value '改h才對
            ElseIf cRow = 0 Then
                Call GetLastClose(Cells(Z, "T").Text)
                If Sheets("TradingDay").Cells(2, "AC").Value <> 0 Then
                    Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 3).Value = Sheets("TradingDay").Cells(2, "AC").Value
                    Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 4).Value = Sheets("TradingDay").Cells(2, "AC").Value
                    Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 5).Value = Sheets("TradingDay").Cells(2, "AC").Value
                    Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 6).Value = Sheets("TradingDay").Cells(2, "AC").Value
                    Sheets("TradingDay").Cells(2, "AC").ClearContents
                Else
                    Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 3).Value = 0
                    Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 4).Value = 0

Sub DisposalSIIMerge()
    Dim strNotice As String
    Dim NoticeArrers() As String
    Dim tDay As Date
    StartRow = Evaluate("MATCH(""證券代號"",V:V,0)+1")
    EndRow = [U1048576].End(xlUp).Row
    Call CheckCsvTitle("DisposalSII", StartRow - 1, CDate(Sheets("MergeData").[S1].Value))
    C = EndRow - StartRow
    m = Range("A1048576").End(xlUp).Offset(1, 0).Row
    e = m + C
    DBDay = DateValue(Sheets("SetUp").Cells(11, "C"))
    For i = StartRow To EndRow
        tDay = Year(CDate(Cells(i, "U"))) + 1911 & "/" & Month(CDate(Cells(i, "U"))) & "/" & Day(CDate(Cells(i, "U")))
        If tDay > DBDay And Len(Trim(Sheets("MergeData").Cells(i, "V").Text)) = 4 Then
            Sheets("MergeData").[A1048576].End(xlUp).Offset(1, 0).Value = tDay '日期
            strCode = CorrectionCode(Trim(Sheets("MergeData").Cells(i, "V").Text)) '股票代號變數
            Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 1).Value = strCode '股票代號"
            Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 2).Value = Trim(Sheets("MergeData").Cells(i, "W").Text) '股票名
            Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 3).Value = "上市"
            '起始日
            siiDate = Mid(Sheets("MergeData").Cells(i, "Z").Text, 1, InStr(Sheets("MergeData").Cells(i, "Z").Text, "~") - 1)
            otcDate = Mid(siiDate, 1, InStr(siiDate, "/") - 1) + 1911 & Mid(siiDate, InStr(siiDate, "/"), 6)
            Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 4).Value = otcDate
            '結束日
            siiDate = Mid(Sheets("MergeData").Cells(i, "Z").Text, InStr(Sheets("MergeData").Cells(i, "Z").Text, "~") + 1, 9)
            otcDate = Mid(siiDate, 1, InStr(siiDate, "/") - 1) + 1911 & Mid(siiDate, InStr(siiDate, "/"), 6)
            Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 5).Value = otcDate
            '處置原因
            Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 6).Value = Trim(Sheets("MergeData").Cells(i, "Y").Text) '處置原因
            '處置方式
            strNotice = Mid(Sheets("MergeData").Cells(i + 3, "T").Text, InStr(Sheets("MergeData").Cells(i + 3, "T").Text, "撮合作業（約每") + 6, InStr(Sheets("MergeData").Cells(i + 3, "T").Text, "分鐘撮合一次") - InStr(Sheets("MergeData").Cells(i + 3, "T").Text, "撮合作業（約每"))
            Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 7).Value = strNotice
            '處置內容
            Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 8).Value = Sheets("MergeData").Cells(i, "AB").Text
        End If
    Next
End Sub

'=====數據清理程式=====
Sub AllMergeData(strSheets As String)
    Dim myFileName As String  ' 設定檔案名稱變數
    Dim strMarket As String '設定市場變數
    Dim MyFile As Object '設定檔案物件變數
    Sheets("MergeData").Select     '作用中的工作表轉移到MergeData
    Columns("B:B").NumberFormatLocal = "@" 'b欄位為文字
    myFileName = Dir(MyPath("csv") & "*") '取得檔案名稱
    Do While myFileName <> "" '檔名不為空白
        If strSheets = "Revenue" Then '如果是月營收
            FileDay = Mid(myFileName, 1, 4) & "/" & Mid(myFileName, 5, 2) & "/" & "01" '取得日期並改格式
        Else '如果不是月營收
            FileDay = Mid(myFileName, 1, 4) & "/" & Mid(myFileName, 5, 2) & "/" & Mid(myFileName, 7, 2) '取得日期並改格式
        End If
        Select Case strSheets
            Case "Close"
                DBDay = DateValue(Sheets("SetUp").Cells(6, "C"))
            Case "Legal"
                DBDay = DateValue(Sheets("SetUp").Cells(7, "C"))
            Case "Margin"
                DBDay = DateValue(Sheets("SetUp").Cells(8, "C"))
            Case "DayTrading"
                DBDay = DateValue(Sheets("SetUp").Cells(9, "C"))
            Case "Revenue"
                DBDay = DateValue(Sheets("SetUp").Cells(18, "C"))
            Case "Notice"
                DBDay = DateValue(Sheets("SetUp").Cells(10, "C"))
            Case "Disposal"
                DBDay = DateValue(Sheets("SetUp").Cells(11, "C"))
        End Select
        Application.StatusBar = "目前進度:" & myFileName '顯示進度
        DoEvents '回應使用者操作事件
        If strSheets = "Revenue" And DateValue(FileDay) > DateValue(DBDay) And DateValue(FileDay) <= DateValue(Sheets("SetUp").Cells(17, "C")) Then GoTo startRevenue '跳到開始處理月營收資料
        If Right(myFileName, 3) = "csv" And Mid(myFileName, 9, 5) = Mid(strSheets, 1, 5) And DateValue(FileDay) > DateValue(DBDay) And DateValue(FileDay) <= DateValue(Sheets("SetUp").Cells(5, "C")) Then ' 判斷檔案格式及日期
startRevenue: '開始處理月營收資料
            strMarket = Mid(Right(myFileName, 7), 1, 3) '取得市場別
            With ActiveSheet.QueryTables.Add("TEXT;" & MyPath("csv") & myFileName, Sheets("MergeData").[$T$1])
                .TextFileParseType = xlDelimited '文件解析類型為使用分隔符分割
                If strSheets = "Revenue" Then .TextFilePlatform = 65001 'UTF-8
                .TextFileTabDelimiter = False '取消tab鍵
                .TextFileCommaDelimiter = True '使用","
                .Refresh BackgroundQuery:=False '是否背景執行
                Sheets("MergeData").Range("S1") = FileDay '輸入FileDay給處理資料的程式使用
                Call SheetsMarket(strSheets & strMarket)
                .ResultRange.Clear '把處理好的資料刪掉
                .Delete '把連線及名稱刪掉
                Sheets("MergeData").Range("S1").ClearContents '刪掉FileDay
            End With
            '移動處理好的csv
            Set MyFile = CreateObject("Scripting.FileSystemObject")
            MyFile.MoveFile MyPath("csv") & myFileName, MyPath("csv") & strSheets & "\"
            Set MyFile = Nothing
        End If
        myFileName = Dir '取得下一個檔名
        If strSheets = "Revenue" And myFileName <> "" Then '如果還有月營收檔案
            FileDay = Mid(myFileName, 1, 4) & "/" & Mid(myFileName, 5, 2) & "/" & "01" '取得日期並改格式
            GoTo startRevenue '繼續處理資料
        ElseIf strSheets = "Revenue" And myFileName = "" Then '如果是最後一個月營收檔案
            GoTo RevenueUpDB '把處理好的資料上傳資料庫
        End If
        If FileDay <> Mid(myFileName, 1, 4) & "/" & Mid(myFileName, 5, 2) & "/" & Mid(myFileName, 7, 2) And DateValue(FileDay) > DateValue(DBDay) And DateValue(FileDay) <= DateValue(Sheets("SetUp").Cells(5, "C")) Then  '如果下一個檔名的日期<>FileDay，就開始處理整理好的資料
RevenueUpDB: '把處理好的資料上傳資料庫
            Call InputTitle(strSheets) '輸入Title
            StartRow = Sheets("MergeData").[A1048576].End(xlUp).Row '取得最後一列
            strColumn = Sheets("MergeData").[A1].End(xlToRight).Column  '取得最後一欄
            EndRow = Sheets("MergeData").Cells(1, CInt(strColumn) - 1).End(xlDown).Row '取得最後一欄的最後一列，檢查用
            If strSheets = "Disposal" And StartRow = 1 Then '如果當天沒資料
                Sheets("SetUp").Select     '作用中的工作表轉移到SetUp
                End '就結束程式
            End If
            If StartRow = EndRow Then
                Sheets.Add After:=Sheets(Sheets.Count)  ' 新增最後1張工作表
                Sheets(Sheets.Count).Select       ' 選取最後一張工作表
                Sheets(Sheets.Count).Name = strSheets '工作表改名
                Columns("B:B").NumberFormatLocal = "@" 'b欄位為文字
                Columns("A:A").NumberFormatLocal = "yyyy-MM-dd" '設定為這個格式，日期才能正確排序
                Sheets(strSheets).[A1].Resize(StartRow, strColumn).Value = Sheets("MergeData").[A1].Resize(StartRow, strColumn).Value '把整理好的資料移到新的工作表
                Sheets(strSheets).Move '移動工作表至新的活頁簿
                '另存為csv_UTF8格式，但副檔名要改成txt，這樣子股票代號才會完整，而且中文股名才會顯示出來
                ActiveWorkbook.SaveAs Filename:="" & MyPath("SQLite") & "All" & strSheets & ".txt", _
                    FileFormat:=xlCSVUTF8, CreateBackup:=False
                ActiveWindow.Close '關掉作用中的活頁簿
                Sheets("MergeData").Select '作用中的工作表轉移到MergeData
                Cells.Select '全選工作表
                Selection.Delete Shift:=xlUp '刪除(不能用ClearContents，有時會留下""資料欄位)
                '上傳資料庫
                fS = FileLen(MyPath("Data") & "Stock.db")  '取得db的容量
                Script = Shell(MyPath("SQLite") & "SQLite.bat " & MyPath("Data") & "stock.db " & strSheets, 0)  '上傳資料庫
                ss = 1
                If strSheets <> "Disposal" Then '因處置資料量少，所以跳過不檢查(或是改成檢查日期，這部份再想想)
                    Do While FileLen(MyPath("Data") & "Stock.db") = fS
                        '如果容量沒改變，代表還沒上傳完成，所以等待1秒再判斷容量是否改變了
                        Application.Wait (Now + TimeValue("0:00:01"))  '設定等待秒數
                        ss = ss + 1
                        If ss = 5 Then MsgBox "ERROR"
                        DoEvents '回應使用者操作事件
                    Loop
                End If
                If strSheets = "Close" Then Sheets("SetUp").Cells(6, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "Legal" Then Sheets("SetUp").Cells(7, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "Margin" Then Sheets("SetUp").Cells(8, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "DayTrading" Then Sheets("SetUp").Cells(9, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "Notice" Then Sheets("SetUp").Cells(10, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                If strSheets = "Disposal" Then
                    Sheets("SetUp").Cells(11, "C") = FileDay '更新資料庫日期(這邊到時要修改)
                    Application.Wait (Now + TimeValue("0:00:01"))  '設定等待秒數，如果是處置資料，這邊要等一秒，資料才會完整
                End If
                If strSheets = "Close" Then Sheets("TradingDay").[A1048576].End(xlUp).Offset(1, 0).Value = FileDay '更新交易日日期(改成只有close要做)
                Kill MyPath("SQLite") & "All" & strSheets & ".txt" '如果a15的值等於空白就刪掉剛剛的檔案"
                If strSheets = "Close" Then Call GetYesterDayData '取得最新資料庫日期的資料
            Else
                MsgBox ("資料有缺，請檢查原始資料")
                End
            End If
        End If
    Loop
    Sheets("SetUp").Select     '作用中的工作表轉移到SetUp
End Sub

'判斷資料及市場並呼叫資料清理程式
Sub SheetsMarket(strSheetsMarket As String)
    Select Case strSheetsMarket
        Case "CloseSII"
            CloseSIIMerge
        Case "CloseOTC"
            CloseOTCMerge
        Case "MarginSII"
            MarginSIIMerge
        Case "MarginOTC"
            MarginOTCMerge
        Case "LegalSII"
            LegalSIIMerge
        Case "LegalOTC"
            LegalOTCMerge
        Case "DayTradingSII", "DayTradingOTC"
            DayTradingMerge
        Case "RevenueSII", "RevenueOTC"
            RevenueMerge
        Case "NoticeSII"
            NoticeSIIMerge
        Case "NoticeOTC"
            NoticeOTCMerge
        Case "DisposalSII"
            DisposalSIIMerge
        Case "DisposalOTC"
            DisposalOTCMerge
    End Select
End Sub

'檢查Csv的標題是否正確
Sub CheckCsvTitle(strSheetsMarket As String, StartRow As Integer, FileDay As Date)
    Dim Ref
    Dim strRef As String
    Select Case strSheetsMarket
        Case "CloseSII"
            Ref = Array("證券代號", "證券名稱", "成交股數", "成交筆數", "成交金額", "開盤價", "最高價", "最低價", "收盤價")
        Case "CloseOTC"
            Ref = Array("代號", "名稱", "收盤", "漲跌", "開盤", "最高", "最低", "均價", "成交股數", "成交金額(元)", "成交筆數")

'月營收
Sub RevenueMerge()
        '選取範圍並復制到B欄
        StartRow = Evaluate("MATCH(""出表日期"",T:T,0)+1")
        EndRow = [W1048576].End(xlUp).Row
        strColumn = Sheets("MergeData").Cells(StartRow - 1, "T").End(xlToRight).Column - 2 '取得最後一欄
        Call CheckCsvTitle("Revenue", StartRow - 1, CDate(Sheets("MergeData").[S1].Value)) '檢查Title
        C = EndRow - StartRow
        m = Range("A1048576").End(xlUp).Offset(1, 0).Row
        e = m + C
        Range("A" & m & ":A" & e).Value = CDate(Sheets("MergeData").[S1].Value) '日期
        Range("B" & m & ":E" & e).Value = Range("V" & StartRow & ":Y" & EndRow).Value '代號、名稱、產業別、當月營收
        Range("F" & m & ":F" & e).Value = Range("AG" & StartRow & ":AG" & EndRow).Value '備註
        For i = m To e
            Sheets("MergeData").Cells(i, "B").Value = CorrectionCode(Trim(Sheets("MergeData").Cells(i, "B").Text)) '處理股票代號
            Sheets("MergeData").Cells(i, "C").Value = Trim(Sheets("MergeData").Cells(i, "C").Text) '處理股票名
        Next
End Sub

Sub DayTradingMerge()
        '選取範圍並復制到B欄
        StartRow = Evaluate("MATCH(""證券代號"",T:T,0)+1")
        EndRow = [W1048576].End(xlUp).Row
        strColumn = Sheets("MergeData").Cells(StartRow - 1, "T").End(xlToRight).Column - 19 '取得最後一欄
        Call CheckCsvTitle("DayTrading", StartRow - 1, CDate(Sheets("MergeData").[S1].Value)) '檢查Title
        C = EndRow - StartRow
        m = Range("A1048576").End(xlUp).Offset(1, 0).Row
        e = m + C
        Range("A" & m & ":A" & e).Value = CDate(Sheets("MergeData").[S1].Value) '日期
        '<= 2014/06/27 otc少一欄，>=2014/06/30都一樣，要處理一下
        If strColumn = 6 Then
            Range("B" & m & ":G" & e).Value = Range("T" & StartRow & ":Y" & EndRow).Value '所有資料
        Else
            Range("B" & m & ":C" & e).Value = Range("T" & StartRow & ":U" & EndRow).Value '所有資料
            Range("E" & m & ":G" & e).Value = Range("V" & StartRow & ":X" & EndRow).Value '所有資料
        End If
        For i = m To e
            Sheets("MergeData").Cells(i, "B").Value = CorrectionCode(Trim(Sheets("MergeData").Cells(i, "B").Text)) '處理股票代號
            Sheets("MergeData").Cells(i, "C").Value = Trim(Sheets("MergeData").Cells(i, "C").Text) '處理股票名
        Next
End Sub

Sub CloseSIIMerge()
    StartRow = Evaluate("MATCH(""證券代號"",T:T,0)+1")
    EndRow = [U1048576].End(xlUp).Row
    Call CheckCsvTitle("CloseSII", StartRow - 1, CDate(Sheets("MergeData").[S1].Value))
    For Z = StartRow To EndRow
        Sheets("MergeData").[A1048576].End(xlUp).Offset(1, 0).Value = CDate(Sheets("MergeData").[S1].Value) '日期
        Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 1).Value = Trim(CStr(Sheets("MergeData").Cells(Z, "T").Text)) '股票代號
        strCode = Cells(Z, "T").Text '股票代號變數
        Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 2).Value = Trim(Sheets("MergeData").Cells(Z, "U").Text) '股票名
        cRow = Code2Row(strCode, "Close")
        If Cells(Z, "Y").Value = "--" Then
            If cRow > 0 Then
                Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 3).Value = Sheets("TradingDay").Cells(cRow, "H").Value '改h才對
                Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 4).Value = Sheets("TradingDay").Cells(cRow, "H").Value '改h才對
                Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 5).Value = Sheets("TradingDay").Cells(cRow, "H").Value '改h才對
                Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 6).Value = Sheets("TradingDay").Cells(cRow, "H").Value '改h才對
            ElseIf cRow = 0 Then
                Call GetLastClose(Cells(Z, "T").Text)
                If Sheets("TradingDay").Cells(2, "AC").Value <> 0 Then
                    Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 3).Value = Sheets("TradingDay").Cells(2, "AC").Value

                    Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 5).Value = 0

            End If
        Else
            Sheets("MergeData").[A1048576].End(xlUp).Offset(0, 3).Value = Cells(Z, "Y").Value * 100

Sub SQLiteDB(strSelect As String, strSheets As String, StartRow As Long, strColumn As String)
    Dim myConn As ADODB.Connection
    Dim myRS As ADODB.Recordset
    Dim i As Long
    Set myConn = New ADODB.Connection
    myConn.Open "driver=SQLite3 ODBC Driver;Database=" & ThisWorkbook.Path & "\Data\stock.db;"
    Set myRS = New ADODB.Recordset
    With myRS
        .CursorLocation = adUseClient
        .Open Source:=strSelect, ActiveConnection:=myConn
    End With
    With ThisWorkbook.Worksheets(strSheets)
'        .Cells.ClearContents'全選刪除
'        .Cells(1, "B").Resize(3000, 11).ClearContents'欄位範圍刪除
        .Cells(StartRow, strColumn).CopyFromRecordset myRS
        '匯入標題例
'        For I = 0 To myRS.Fields.Count - 1
'            .Cells(1, I + 1) = myRS.Fields(I).Name
'        Next I
    End With
    myRS.Close
    myConn.Close
    Set myRS = Nothing
    Set myConn = Nothing
End Sub

    '取得最新交易日資料

Sub VersionUpdate()
    Dim strUpdate As String
    i = 1
    Do While Sheets("VersionUpdate").Cells(i, "B") <> "" '檔名不為空白
    If Sheets("VersionUpdate").Cells(i, "A") <> "" Then
        strUpdate = strUpdate & Sheets("VersionUpdate").Cells(i, "A") & vbTab & Sheets("VersionUpdate").Cells(i, "B") & vbCrLf
    Else
        strUpdate = strUpdate & vbTab & Sheets("VersionUpdate").Cells(i, "B") & vbCrLf
    End If
    i = i + 1
    Loop
    Debug.Print strUpdate
    MsgBox strUpdate, vbInformation, "=====更新紀錄====="
End Sub
